# Ion test 4 (view counts; first written 2026-11-21, as a .txt file 2026-11-25): the view count of one file, on
# usaf.dps.mil only. Read-only, one site, one file. Prints only counts, True/False and error type names: no address,
# no file name and no error message. No Microsoft Graph call: the addresses are on usaf.dps.mil only.
$ErrorActionPreference = 'Stop'
'T4 test version=2026-11-25c'
$typed = (Read-Host 'Question 1: the address of the folder page in your web browser').Trim().Trim('"')
$fileText = (Read-Host 'Question 2: the file name with its type (for example Report.pdf)').Trim().Trim('"', '/')
Clear-Host
'T4 test version=2026-11-25c'
$tag = 'T4'
$signedIn = $false
try {
    function Get-Kind($err) { $err.Exception.GetType().Name }   # the error type name only, never the message
    if ($typed -notmatch '^https?://') { $typed = 'https://' + $typed }
    $siteMatch = [regex]::Match($typed, '^https://usaf\.dps\.mil/((?:sites|teams)/[^/?#\s]+)')
    if (-not $siteMatch.Success) { 'T4 answer 1 is a DoD365 site address=False'; return }   # 'finally' prints Done.
    if (-not $fileText) { 'T4 answer 2 given=False'; return }
    $site = 'https://usaf.dps.mil/' + $siteMatch.Groups[1].Value
    # The folder of the file, from the address of the folder page in the web browser: its 'id=' part (a folder), or the
    # part before '/Forms/' (the library), or the path of the address. Only a folder of this one site is used.
    $sitePath = '/' + $siteMatch.Groups[1].Value
    $u = [uri]$typed
    $folderPath = ''; $folderFrom = 'none'
    $q = [regex]::Match($u.Query, '(?i)(?:^\?|&)(?:id|RootFolder)=([^&]+)')
    if ($q.Success) { $folderPath = [uri]::UnescapeDataString($q.Groups[1].Value); $folderFrom = 'id' }
    else {
        $p = [uri]::UnescapeDataString($u.AbsolutePath).TrimEnd('/')
        $i = $p.IndexOf('/Forms/', [StringComparison]::OrdinalIgnoreCase)
        if ($i -gt 0) { $folderPath = $p.Substring(0, $i); $folderFrom = 'library page' }
        elseif ($p.Length -gt $sitePath.Length) { $folderPath = $p; $folderFrom = 'path' }
    }
    $folderPath = $folderPath.TrimEnd('/')
    $name = [uri]::UnescapeDataString($fileText)
    if (($folderPath + '/' + $name) -match '(^|/)\.\.(/|$)') { '{0} answer with .. found=True: stopped' -f $tag; return }
    if ($name.Contains('/')) { $folderFrom = 'answer 2 (a path)'; $path = $sitePath + '/' + $name.TrimStart('/') }
    elseif (-not $folderPath.StartsWith($sitePath + '/', [StringComparison]::OrdinalIgnoreCase)) { '{0} folder in the address found=False' -f $tag; return }
    else { $path = $folderPath + '/' + $name }
    '{0} folder from={1}' -f $tag, $folderFrom
    $signedIn = $true   # before the sign-in: a stop during it still signs out
    $clock = [Diagnostics.Stopwatch]::StartNew()
    Connect-PnPOnline -Url $site -UseWebLogin -WarningAction SilentlyContinue
    'T4 sign-in ok=True ms={0}' -f $clock.ElapsedMilliseconds
    $clock.Restart()
    $item = $null; $by = 'name'
    # A file that is not there gives nothing (PnP 1.12.0) or an error: both go on to the search below.
    try { $item = Get-PnPFile -Url $path -AsListItem } catch { $item = $null }
    # The web page can show a name without its type: only then, the files of that one folder by the name without its type.
    if (-not $item -and $folderFrom -ne 'answer 2 (a path)' -and -not [IO.Path]::GetExtension($name)) {
        try {
            $inFolder = Get-PnPFolderItem -FolderSiteRelativeUrl $folderPath.Substring($sitePath.Length + 1) -ItemType File
            $same = @(foreach ($one in $inFolder) { $one }) | Where-Object { [IO.Path]::GetFileNameWithoutExtension($_.Name) -ieq $name }
            $same = @($same)
            if ($same.Count -eq 1) { $item = Get-PnPFile -Url $same[0].ServerRelativeUrl -AsListItem; $by = 'name without its type' }
            else { 'T4 files with this name in the folder={0}' -f $same.Count }
        }
        catch { 'T4 folder read ok=False {0}' -f (Get-Kind $_) }
    }
    'T4 file found={0} by={1} ms={2}' -f [bool]$item, $by, $clock.ElapsedMilliseconds
    if (-not $item) { return }
    $list = Get-PnPProperty -ClientObject $item -Property ParentList
    $siteId = (Get-PnPSite -Includes Id).Id; $webId = (Get-PnPWeb -Includes Id).Id
    $address = '/_api/v2.1/sites/usaf.dps.mil,{0},{1}/lists/{2}/items/{3}' -f $siteId, $webId, $list.Id, $item.Id
    # Method 1: all-time count (the SharePoint item analytics, as Graph's itemAnalytics).
    try {
        $all = Invoke-PnPSPRestMethod -Method Get -Url ($address + '/analytics/allTime')
        $count = ''; if ($all -and $all.PSObject.Properties['access']) { $count = [string]$all.access.actionCount }
        if ($count -ne '') { 'T4a works: {0} views, all time' -f $count } else { 'T4a no answer' }
    }
    catch { 'T4a failed: {0}' -f $_.Exception.GetType().Name }
    # Method 2: views per month from a start date (what Ion needs: the views after the "PII first saved" date).
    try {
        $start = (Get-Date).AddDays(-89).ToString('yyyy-MM-dd'); $end = (Get-Date).ToString('yyyy-MM-dd')
        $url = "$address/getActivitiesByInterval(startDateTime='$start',endDateTime='$end',interval='month')"
        $months = Invoke-PnPSPRestMethod -Method Get -Url $url
        $total = 0; $rows = 0
        if ($months -and $months.PSObject.Properties['value']) {
            foreach ($month in @($months.value)) { $rows++; if ($month.access) { $total += [int]$month.access.actionCount } }
        }
        if ($rows -gt 0) { 'T4b works: {0} views in the last 90 days, {1} months' -f $total, $rows } else { 'T4b no answer' }
    }
    catch { 'T4b failed: {0}' -f $_.Exception.GetType().Name }
}
catch { 'T4 stopped: {0}' -f $_.Exception.GetType().Name }
finally {
    if ($signedIn) { try { Disconnect-PnPOnline } catch { Write-Verbose 'Already signed out.' } }
    Write-Host 'Done. The lines above show counts, True/False and error type names only. Copy them into an e-mail to yourself.'
}
